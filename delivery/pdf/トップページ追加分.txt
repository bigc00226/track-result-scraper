// 選んだ PDF は新しいタブの画面に直接渡します（サーバには送信しません）
function pdfToolGo(form, page, mode) {
    var f = form.pdf.files && form.pdf.files[0];
    if (!f) { alert('PDFファイルを選択してください'); return false; }
    window.__pdfToolFile = f;
    window.__pdfToolMode = mode;
    window.open(page, '_blank');
    return false;
}
</script>
